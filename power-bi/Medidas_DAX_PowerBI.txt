-- INSURANCE INTELLIGENCE | MEDIDAS DAX INICIAIS
-- Modelo demonstrativo com dados sintéticos.
-- Relacionamentos recomendados:
-- DimMes[AnoMes] 1:* FatoPropostas[AnoMes]
-- DimMes[AnoMes] 1:* FatoApolices[AnoMes]
-- DimMes[AnoMes] 1:* FatoRenovacoes[AnoMes]
-- DimMes[AnoMes] 1:* Metas[AnoMes]
-- DimProduto[ProdutoID] 1:* fatos e Metas[ProdutoID]
-- DimEquipe[EquipeID] 1:* fatos correspondentes.
-- DimUnidade[UnidadeID] 1:* fatos correspondentes.
-- Direção de filtro: dimensões -> fatos.
--
-- A conversão abaixo é calculada pela coorte da proposta:
-- propostas emitidas / propostas registradas no período de origem.
-- Não divide apólices emitidas no mês por propostas abertas no mesmo mês.

Propostas = COUNTROWS(FatoPropostas)

Propostas Convertidas =
CALCULATE(
    COUNTROWS(FatoPropostas),
    FatoPropostas[StatusProposta] = "Emitida"
)

Apolices Emitidas = COUNTROWS(FatoApolices)

Premio Emitido = SUM(FatoApolices[PremioEmitido])

Premio Medio = DIVIDE([Premio Emitido], [Apolices Emitidas])

Taxa Conversao = DIVIDE([Propostas Convertidas], [Propostas])

Renovacoes Registradas = COUNTROWS(FatoRenovacoes)

Renovacoes Concluidas =
CALCULATE(
    COUNTROWS(FatoRenovacoes),
    FatoRenovacoes[StatusRenovacao] = "Renovada"
)

Taxa Renovacao = DIVIDE([Renovacoes Concluidas], [Renovacoes Registradas])

Renovacoes Pendentes =
CALCULATE(
    COUNTROWS(FatoRenovacoes),
    FatoRenovacoes[StatusRenovacao] = "Pendente"
)

Cancelamentos Renovacao =
CALCULATE(
    COUNTROWS(FatoRenovacoes),
    FatoRenovacoes[StatusRenovacao] = "Cancelada"
)

Meta Premio = SUM(Metas[MetaPremio])

Atingimento Meta = DIVIDE([Premio Emitido], [Meta Premio])

Gap Meta = [Premio Emitido] - [Meta Premio]
